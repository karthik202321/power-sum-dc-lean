do not need this wrapper. The wrapper source is included below for review.

Compatibility wrapper source:

#define _GNU_SOURCE
#include <dlfcn.h>
#include <stdio.h>
#include <string.h>
#include <unistd.h>
/* Lean 4.19 uses /proc/<pid>/exe. This environment exposes /proc/self/exe. */
ssize_t readlink(const char *path, char *buf, size_t size) {
 static ssize_t (*original)(const char *, char *, size_t);
 if (!original) original = dlsym(RTLD_NEXT, "readlink");
 char own[64]; snprintf(own, sizeof own, "/proc/%d/exe", (int)getpid());
 return original(strcmp(path, own) == 0 ? "/proc/self/exe" : path, buf, size);
}
