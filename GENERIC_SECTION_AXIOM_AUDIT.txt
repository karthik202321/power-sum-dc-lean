Sheshadri.ActualGenericGaussSection.exists_actual_section.{u_1, u_2} {n m : ℕ} {D : Type u_1} {L : Type u_2}
  [CommRing D] [IsDomain D] [Algebra ℂ D] [Field L] [Algebra ℂ L]
  (family : Sheshadri.DeterminantDvrFamily.Family n m D L) (hn : 2 ≤ n)
  (s : Sheshadri.ConormalGaussSection.Section (Sheshadri.PolynomialModel.fermat n) n) :
  ∃ ell a J c,
    (∀ (j : ℕ) (i : Fin n), family.residue (ell j i) = s.ell j i) ∧
      (∀ (i : Fin n), family.residue (a i) = s.a i) ∧
        Module.Finite L (MvPolynomial (Fin n) L ⧸ J) ∧

              IsUnit
                  ((Ideal.Quotient.mk J)
                    (∑ i,
                      MvPolynomial.C (c i) *
                        (MvPolynomial.pderiv i) (Sheshadri.PolynomialModel.determinant family.pencil))) ∧
                J =
